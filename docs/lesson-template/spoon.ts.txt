/** What a number of spoons means for the bowls: one each, some bowls with none, or spoons left over. */
export type Meaning = { kind: 'full' } | { kind: 'short'; missing: number } | { kind: 'over'; extra: number };
export function meaning(spoons: number, bowls: number): Meaning {
  return spoons === bowls ? { kind: 'full' } : spoons < bowls ? { kind: 'short', missing: bowls - spoons } : { kind: 'over', extra: spoons - bowls };
}
